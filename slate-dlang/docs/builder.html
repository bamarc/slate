slate.agent.builder

Agent assembly: builder, protocol-adapter slots, and lifecycle handle.

Discussion

Port of slate_agent::builder: AgentBuilder wiring (default Slint backends or custom overrides, pluggable protocol adapters) returning an AgentHandle. Adaptations:

  • Rust is generic over the Slint window type and leaks a tokio runtime (ensure_tokio_runtime). D threads start on demand, so no runtime exists; the builder wires engine components + bridge + adapters.
  • Slint-backed inspector/controller/renderer need element-query APIs that slint-d does not expose yet (slate.agent.slint_backend, next); until then the builder composes caller-provided or default components, exactly the seam Rust's with_inspector/... offers.
  • Heterogeneous adapters erase to a private slot interface capturing the typed attach call (the canonical D substitute for storing distinct AgentProtocolAdapter instantiations).

    Examples

    1. import slate.agent.engine : DefaultElementController, DefaultStateProvider,
    DefaultUiInspector, DefaultVisualRenderer;
import slate.agent.protocol : ProtocolServerHandle;
import std.exception : assertThrown;
import std.typecons : Nullable;

static class ScriptHandle : ProtocolServerHandle
{
    bool running = true;
    bool stopped;
    string[] log;
    override bool isRunning()
    {
        return running && !stopped;
    }

    override void stop()
    {
        stopped = true;
    }

    override Nullable!string localAddr()
    {
        return Nullable!string.init;
    }
}

static class ScriptAdapter : AgentProtocolAdapter!(string, ScriptHandle)
{
    ScriptHandle last;
    override ScriptHandle attach(AutomationEngine engine, string config)
    {
        assert(engine !is null && config == "test");
        last = new ScriptHandle();
        return last;
    }
}

// Missing components fail with the engine builder's messages.
assertThrown!AgentError(new AgentBuilder().build());

auto adapter = new ScriptAdapter();
auto handle = new AgentBuilder()
    .withName("demo")
    .withInspector(new DefaultUiInspector())
    .withController(new DefaultElementController())
    .withStateProvider(new DefaultStateProvider())
    .withRenderer(new DefaultVisualRenderer())
    .withProtocolAdapter!(ScriptAdapter, string, ScriptHandle)(adapter, "test")
    .start();

assert(handle.name == "demo" && handle.isRunning());
assert(handle.serverHandles().length == 1);
assert(handle.engine().inspector().inspectTree().countElements() == 1);
handle.stop();
assert(!handle.isRunning());
handle.shutdown(); // idempotent
